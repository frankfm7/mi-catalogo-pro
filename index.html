📄 sw.js 
const CACHE_NAME = 'catalogo-pro-v1.0';
const urlsToCache = [
  '/',
  '/app.html',
  '/app.js',
  '/styles.css',
  '/index.html'
];

self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(urlsToCache))
  );
});

self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request)
      .then(response => response || fetch(event.request))
  );
});

const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

const state = {
  currentUser: null,
  catalogs: [],
  products: [],
  sales: [],
  currentView: 'catalogs',
  isOnline: navigator.onLine
};

document.addEventListener('DOMContentLoaded', () => {
  console.log('✅ Catálogo Pro v1.0 iniciado');
  lucide.createIcons();
  checkSession();
  setupEventListeners();
});

async function checkSession() {
  const { data: { session } } = await supabase.auth.getSession();
  if (session) {
    state.currentUser = session.user;
    showMainScreen();
    loadData();
  } else {
    showLoginScreen();
  }
}

function setupEventListeners() {
  document.getElementById('enter-btn')?.addEventListener('click', () => {
    document.getElementById('welcome-screen').classList.add('hidden');
    document.getElementById('login-screen').classList.remove('hidden');
  });

  document.getElementById('login-form')?.addEventListener('submit', handleLogin);
  document.getElementById('menu-btn')?.addEventListener('click', toggleSidebar);
}

async function handleLogin(e) {
  e.preventDefault();
  const username = document.getElementById('login-username').value;
  const password = document.getElementById('login-password').value;

  try {
    const { data, error } = await supabase.auth.signInWithPassword({
      email: username,
      password: password
    });

    if (error) throw error;
    state.currentUser = data.user;
    showMainScreen();
    loadData();
  } catch (error) {
    alert('Error: ' + error.message);
  }
}

function showMainScreen() {
  document.getElementById('login-screen').classList.add('hidden');
  document.getElementById('main-screen').classList.remove('hidden');
}

function showLoginScreen() {
  document.getElementById('welcome-screen').classList.add('hidden');
  document.getElementById('login-screen').classList.remove('hidden');
}

function toggleSidebar() {
  document.getElementById('sidebar').classList.toggle('open');
}

async function loadData() {
  try {
    const { data: catalogs } = await supabase.from('catalogs').select('*');
    state.catalogs = catalogs || [];
    renderCatalogs();
  } catch (error) {
    console.error('Error:', error);
  }
}

function renderCatalogs() {
  const grid = document.getElementById('catalogs-grid');
  if (!grid) return;

  grid.innerHTML = '';
  state.catalogs.forEach(catalog => {
    const card = document.createElement('div');
    card.className = 'catalog-card';
    card.innerHTML = `
      

        
${catalog.name}

      

    `;
    grid.appendChild(card);
  });
}

function showToast(message) {
  alert(message);
}
